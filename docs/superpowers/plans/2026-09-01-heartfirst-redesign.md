# HeartFirst Surat Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the HeartFirst Surat site in a new "The Index" editorial design — every page's content taken verbatim from the eight supplied source pages, fully responsive, statically prerendered with per-page SEO.

**Architecture:** Vite + React 19 + TypeScript SPA using `react-router` v7. Page content lives in typed data modules under `src/content/` (transcribed from `src/content/*.html`); route components render that data through a shared layout (numbered left rail, page masthead, contact bar, footer). A custom Vite SSR build + prerender script writes a static HTML file per route so crawlers and no-JS clients get full content. A small `Seo` component collects `<title>`/meta/OpenGraph/JSON-LD during SSR and updates them on the client. Styling is Tailwind v4 utilities plus a CSS `@theme` token layer.

**Tech Stack:** Vite 8, React 19, TypeScript, `react-router` v7, Tailwind CSS v4 (`@tailwindcss/vite`), `lucide-react`, Vitest (added), `linkedom` (added, prerender-time HTML parsing for the extraction scripts), pnpm.

## Conventions for every task

- Package manager is **pnpm**. Run commands from repo root.
- Commit after each task with the message shown. Never `git push`.
- Tailwind v4: no `tailwind.config.js`; theme tokens are declared in `src/styles/index.css` with `@theme`.
- Colour tokens: `--color-indigo: #3f51b5`, `--color-indigo-deep: #31408f`, `--color-yellow: #ffea00`, `--color-ink: #1a1c2b`, `--color-mute: #5c5e6d`, `--color-line: #dcdce4`, `--color-paper: #ffffff`. Use Tailwind classes `text-indigo`, `bg-indigo`, `border-line`, etc.
- Fonts: `Fraunces` (display, `font-display`), `Inter` (UI, default `font-sans`), `Newsreader` (long-form, `font-serif`). Loaded via Google Fonts `@import` at the top of `index.css`.
- Never use `dangerouslySetInnerHTML`. Never leave an `<img>` without `alt`.
- All content strings come from the source pages; see the spec (`docs/superpowers/specs/2026-09-01-heartfirst-redesign-design.md`) §3 for the per-page inventory.

```js
import { readFileSync, writeFileSync } from 'node:fs'
import { parseHTML } from 'linkedom'

const src = readFileSync('src/content/scientific-publications.html', 'utf8')
const { document } = parseHTML(src)

// The publications are the <li> items inside the ONE <ol> on this page.
const ol = document.querySelector('section.scientific-publication-content ol')
  || document.querySelector('ol')
const items = [...ol.querySelectorAll(':scope > li')].map((li, i) => ({
  n: i + 1,

  `import type { Publication } from './types'\n\n` +
  `// Generated by scripts/extract-publications.mjs from the source page — do not hand-edit.\n` +
  `export const publications: Publication[] = ${JSON.stringify(items, null, 2)}\n`

writeFileSync('src/content/publications.ts', body)
console.log(`wrote ${items.length} publications`)
```

## Task 5: Extract the multicentre trials table

**Files:**
- Create: `scripts/extract-trials.mjs`, `src/content/trials.ts` (generated)
- Test: `src/content/__tests__/trials.test.ts`

- [ ] **Step 1: Write the extraction script**

`scripts/extract-trials.mjs`:

```js
import { readFileSync, writeFileSync } from 'node:fs'
import { parseHTML } from 'linkedom'

const src = readFileSync('src/content/career-Hightlight.html', 'utf8')
const { document } = parseHTML(src)

// Find the heading, then the next <table> after it.
const heading = [...document.querySelectorAll('h4')]
  .find(h => /Multicentre National/i.test(h.textContent))
let el = heading
while (el && el.tagName !== 'TABLE') el = el.nextElementSibling
  || (el.parentElement && el.parentElement.nextElementSibling)
const table = el && el.tagName === 'TABLE' ? el : document.querySelector('table')

const rows = [...table.querySelectorAll('tbody tr')]
  .map(tr => [...tr.querySelectorAll('td')].map(td => td.textContent.replace(/\s+/g, ' ').trim()))
  .filter(cells => cells.length >= 2 && cells[0])
  .map(([molecule, condition = '', sponsor = '', cro = '']) => ({ molecule, condition, sponsor, cro }))

if (rows.length < 15) throw new Error(`only found ${rows.length} trial rows`)

writeFileSync(
  'src/content/trials.ts',
  `import type { Trial } from './types'\n\n` +
  `// Generated by scripts/extract-trials.mjs — do not hand-edit.\n` +
  `export const trials: Trial[] = ${JSON.stringify(rows, null, 2)}\n`,
)
console.log(`wrote ${rows.length} trials`)
```

```tsx
import { createContext, useContext, useEffect } from 'react'

export type SeoSink = { tags: string[] }
export type SeoProps = {
  title: string
  description: string
  canonical: string
  jsonLd?: object
}

const SeoContext = createContext<SeoSink | null>(null)
export function SeoProvider({ sink, children }: { sink: SeoSink; children: React.ReactNode }) {
  return <SeoContext.Provider value={sink}>{children}</SeoContext.Provider>
}

const SITE = 'https://heartfirstsurat.com'
const OG_IMAGE = `${SITE}/og-cover.png`

function tagsFor({ title, description, canonical, jsonLd }: SeoProps): string[] {
  const url = SITE + canonical
  const t = [
    `<title>${esc(title)}</title>`,
    `<meta name="description" content="${esc(description)}"/>`,
    `<link rel="canonical" href="${url}"/>`,
    `<meta property="og:type" content="website"/>`,
    `<meta property="og:title" content="${esc(title)}"/>`,
    `<meta property="og:description" content="${esc(description)}"/>`,
    `<meta property="og:url" content="${url}"/>`,
    `<meta property="og:image" content="${OG_IMAGE}"/>`,
    `<meta name="twitter:card" content="summary_large_image"/>`,
  ]
  if (jsonLd) t.push(`<script type="application/ld+json">${JSON.stringify(jsonLd)}</script>`)
  return t
}

function esc(s: string) {
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;')
}

export function Seo(props: SeoProps) {
  const sink = useContext(SeoContext)
  if (sink) sink.tags.push(...tagsFor(props)) // SSR path

  useEffect(() => {
    document.title = props.title
    setMeta('name', 'description', props.description)
    setLink('canonical', SITE + props.canonical)
    setMeta('property', 'og:title', props.title)
    setMeta('property', 'og:description', props.description)
    setMeta('property', 'og:url', SITE + props.canonical)
    setJsonLd(props.jsonLd)
  }, [props.title, props.description, props.canonical, props.jsonLd])

  return null
}

function setMeta(attr: 'name' | 'property', key: string, value: string) {
  let el = document.head.querySelector<HTMLMetaElement>(`meta[${attr}="${key}"]`)
  if (!el) { el = document.createElement('meta'); el.setAttribute(attr, key); document.head.appendChild(el) }
  el.setAttribute('content', value)
}
function setLink(rel: string, href: string) {
  let el = document.head.querySelector<HTMLLinkElement>(`link[rel="${rel}"]`)
  if (!el) { el = document.createElement('link'); el.rel = rel; document.head.appendChild(el) }
  el.href = href
}
function setJsonLd(obj?: object) {
  document.head.querySelector('script[type="application/ld+json"][data-dynamic]')?.remove()
  if (!obj) return
  const s = document.createElement('script')
  s.type = 'application/ld+json'
  s.dataset.dynamic = 'true'
  s.textContent = JSON.stringify(obj)
  document.head.appendChild(s)
}
```

```tsx
import { StrictMode } from 'react'
import { renderToString } from 'react-dom/server'
import { StaticRouter } from 'react-router'
import App from './App'

export async function render(url: string): Promise<{ html: string; head: string }> {
```

```js
  const { html, head } = await render(url)
```
